# Lifeline backend dependencies
# Kept minimal on purpose. The deterministic calculator and the data
# models use only the Python standard library, so they run and test with
# zero third-party packages. The extras below are only needed when you
# actually talk to AWS or serve the HTTP API.

# AWS SDK — only used by the Bedrock service module. The calculator and
# models do NOT import this, so tests run without it.
boto3>=1.34.0

# Loads backend/.env into environment variables for local development.
python-dotenv>=1.0.0

# Test runner.
pytest>=8.0.0

# Gemini Live SDK — the real-time voice layer. Used ONLY by
# app/gemini_service.py to mint short-lived Live tokens server-side.
# The browser never receives the API key. Gemini handles speech in and out;
# every calculation still comes from calculator.py and every piece of
# insurance education still comes from the Bedrock Knowledge Base.
#
# NOTE for Lambda: this and its dependencies (pydantic, websockets, httpx,
# cryptography...) are NOT provided by the Lambda runtime. backend/deploy.ps1
# vendors them into lambda_pkg.zip. Installing here alone is not enough to
# deploy — the deploy script must run under this same interpreter.
google-genai>=2.28.0
